// === Data === const mu = 0.057; const sigma = 0.362; const dataPoints = [-0.4, -0.2, -0.1, 0.1, 0.2, 0.74]; // === Histogram bins (width 0.3mm) === const binWidth = 0.3; const binEdges = [-0.6, -0.3, 0.0, 0.3, 0.6, 0.9]; const bins = []; for (let i = 0; i < binEdges.length - 1; i++) { const lo = binEdges[i], hi = binEdges[i+1]; const count = dataPoints.filter(d => d >= lo && d < hi).length; const density = count / (dataPoints.length * binWidth); bins.push({ lo, hi, center: (lo+hi)/2, density }); } // === Layout === const W = canvas.width, H = canvas.height; const pad = { left: 70, right: 40, top: 70, bottom: 80 }; const gW = W - pad.left - pad.right; const gH = H - pad.top - pad.bottom; const xMin = mu - 3.8 * sigma; const xMax = mu + 3.8 * sigma; const yMax = 1.35 / (sigma * Math.sqrt(2 * Math.PI)); function toCanvasX(x) { return pad.left + (x - xMin) / (xMax - xMin) * gW; } function toCanvasY(y) { return pad.top + gH - (y / yMax) * gH; } function gaussian(x) { return Math.exp(-0.5 * ((x - mu) / sigma) ** 2) / (sigma * Math.sqrt(2 * Math.PI)); } // === Background === ctx.fillStyle = '#1a1a2e'; ctx.fillRect(0, 0, W, H); // === Title === ctx.fillStyle = 'white'; ctx.font = 'bold 17px Arial'; ctx.textAlign = 'center'; ctx.fillText(`PHÂN PHỐI CHUẨN SAI SỐ ODOM (μ = ${mu.toFixed(3)} mm, σ = ${sigma.toFixed(3)} mm)`, W/2, 32); // === Fill regions === const regions = [ { a: -3.8, b: -3, color: '#1a5c30' }, { a: -3, b: -2, color: '#2d8a4e' }, { a: -2, b: -1, color: '#b05a7a' }, { a: -1, b: 0, color: '#3a5fc8' }, { a: 0, b: 1, color: '#3a5fc8' }, { a: 1, b: 2, color: '#b05a7a' }, { a: 2, b: 3, color: '#2d8a4e' }, { a: 3, b: 3.8, color: '#1a5c30' }, ]; for (const r of regions) { const steps = 300; const xa = mu + r.a * sigma, xb = mu + r.b * sigma; ctx.beginPath(); ctx.moveTo(toCanvasX(xa), toCanvasY(0)); for (let i = 0; i <= steps; i++) { const xi = xa + (xb - xa) * i / steps; ctx.lineTo(toCanvasX(xi), toCanvasY(gaussian(xi))); } ctx.lineTo(toCanvasX(xb), toCanvasY(0)); ctx.closePath(); ctx.fillStyle = r.color; ctx.fill(); } // === Gaussian curve === ctx.beginPath(); const steps = 800; for (let i = 0; i <= steps; i++) { const xi = xMin + (xMax - xMin) * i / steps; const cy = toCanvasY(gaussian(xi)); i === 0 ? ctx.moveTo(toCanvasX(xi), cy) : ctx.lineTo(toCanvasX(xi), cy); } ctx.strokeStyle = 'white'; ctx.lineWidth = 2.5; ctx.stroke(); // === Baseline === ctx.beginPath(); ctx.moveTo(pad.left, toCanvasY(0)); ctx.lineTo(pad.left + gW, toCanvasY(0)); ctx.strokeStyle = 'rgba(255,255,255,0.4)'; ctx.lineWidth = 1.5; ctx.stroke(); // === Sigma vertical lines & labels === const sigmaMarks = [-3, -2, -1, 0, 1, 2, 3]; const sigmaNames = ['-3σ', '-2σ', '-1σ', 'μ', '+1σ', '+2σ', '+3σ']; for (let i = 0; i < sigmaMarks.length; i++) { const xv = mu + sigmaMarks[i] * sigma; const cx = toCanvasX(xv); ctx.beginPath(); ctx.moveTo(cx, toCanvasY(gaussian(xv))); ctx.lineTo(cx, toCanvasY(0) + 10); ctx.strokeStyle = 'rgba(255,255,255,0.8)'; ctx.lineWidth = sigmaMarks[i] === 0 ? 2 : 1.2; ctx.setLineDash(sigmaMarks[i] === 0 ? [] : []); ctx.stroke(); // σ label ctx.fillStyle = 'white'; ctx.font = 'bold 12px Arial'; ctx.textAlign = 'center'; ctx.fillText(sigmaNames[i], cx, toCanvasY(0) + 26); // mm value ctx.fillStyle = '#aaaaaa'; ctx.font = '10px Arial'; ctx.fillText(xv.toFixed(2), cx, toCanvasY(0) + 42); } // === Percentage labels === const pctLabels = [ { sx: 0.5, txt: '34.1%', sy: 0.45 }, { sx: -0.5, txt: '34.1%', sy: 0.45 }, { sx: 1.5, txt: '13.6%', sy: 0.20 }, { sx: -1.5, txt: '13.6%', sy: 0.20 }, { sx: 2.5, txt: '2.1%', sy: 0.06 }, { sx: -2.5, txt: '2.1%', sy: 0.06 }, { sx: 3.3, txt: '0.1%', sy: 0.015 }, { sx: -3.3, txt: '0.1%', sy: 0.015 }, ]; for (const p of pctLabels) { const xv = mu + p.sx * sigma; const yv = gaussian(mu) * p.sy; ctx.fillStyle = 'white'; ctx.font = 'bold 12px Arial'; ctx.textAlign = 'center'; ctx.fillText(p.txt, toCanvasX(xv), toCanvasY(yv)); } // === Histogram bars === for (const bin of bins) { if (bin.density === 0) continue; const bx1 = toCanvasX(bin.lo); const bx2 = toCanvasX(bin.hi); const by = toCanvasY(bin.density); const by0 = toCanvasY(0); // Fill ctx.fillStyle = 'rgba(255, 221, 0, 0.30)'; ctx.fillRect(bx1 + 1, by, bx2 - bx1 - 2, by0 - by); // Border ctx.strokeStyle = '#ffdd00'; ctx.lineWidth = 2; ctx.setLineDash([]); ctx.strokeRect(bx1 + 1, by, bx2 - bx1 - 2, by0 - by); // Density label on bar ctx.fillStyle = '#ffdd00'; ctx.font = 'bold 11px Arial'; ctx.textAlign = 'center'; ctx.fillText(bin.density.toFixed(2), toCanvasX(bin.center), by - 6); } // === Individual data points (dots on x-axis) === for (const dp of dataPoints) { const cx = toCanvasX(dp); const cy = toCanvasY(0) + 6; ctx.beginPath(); ctx.arc(cx, cy + 6, 5, 0, 2 * Math.PI); ctx.fillStyle = '#ff6600'; ctx.fill(); ctx.strokeStyle = 'white'; ctx.lineWidth = 1; ctx.stroke(); ctx.fillStyle = '#ff9944'; ctx.font = '9px Arial'; ctx.textAlign = 'center'; ctx.fillText(dp.toFixed(2), cx, cy + 24); } // === Legend === ctx.beginPath(); ctx.arc(pad.left + 10, toCanvasY(0) + 14, 5, 0, 2 * Math.PI); ctx.fillStyle = '#ff6600'; ctx.fill(); ctx.strokeStyle = 'white'; ctx.lineWidth = 1; ctx.stroke(); ctx.fillStyle = '#ff9944'; ctx.font = 'italic 11px Arial'; ctx.textAlign = 'left'; ctx.fillText(' = điểm đo thực tế', pad.left + 18, toCanvasY(0) + 18); ctx.fillStyle = 'rgba(255,221,0,0.5)'; ctx.fillRect(pad.left + 140, toCanvasY(0) + 7, 14, 14); ctx.strokeStyle = '#ffdd00'; ctx.lineWidth = 1.5; ctx.strokeRect(pad.left + 140, toCanvasY(0) + 7, 14, 14); ctx.fillStyle = '#ffdd00'; ctx.font = 'italic 11px Arial'; ctx.textAlign = 'left'; ctx.fillText(' = histogram mật độ', pad.left + 157, toCanvasY(0) + 18); // === X-axis label === ctx.fillStyle = '#cccccc'; ctx.font = '13px Arial'; ctx.textAlign = 'center'; ctx.fillText('Sai số (mm)', W / 2, H - 8); // === Color legend === const legendItems = [ { color: '#3a5fc8', txt: '±1σ → 68.2%' }, { color: '#b05a7a', txt: '±2σ → 95.4%' }, { color: '#2d8a4e', txt: '±3σ → 99.7%' }, ]; let lx = W - 200, ly = pad.top + 10; for (const item of legendItems) { ctx.fillStyle = item.color; ctx.fillRect(lx, ly, 18, 14); ctx.fillStyle = 'white'; ctx.font = '11px Arial'; ctx.textAlign = 'left'; ctx.fillText(item.txt, lx + 24, ly + 11); ly += 22; }
